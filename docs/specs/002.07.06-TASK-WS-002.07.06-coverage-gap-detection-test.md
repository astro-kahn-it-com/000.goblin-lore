  cat << 'EOF' > worker-sower/scripts/test-coverage-gap-detection.sh
  #!/usr/bin/env bash
  set -euo pipefail
  # Temporarily add stub mechanic to registry assertion check
  if node -e "
    const { REGISTERED_RELEASE_MECHANICS } = require('./worker-sower/test/residueCoverageRegistry');
    if (!REGISTERED_RELEASE_MECHANICS.find(m => m.mechanicName === 'UnregisteredMechanic')) {

      process.exit(0);
    }

  fi
